Полный план реорганизации и доработки проекта IP-CSS
Введение
IP-CSS (IP Camera Surveillance System) — кроссплатформенная система видеонаблюдения с веб-интерфейсом, AI-аналитикой, поддержкой IP-камер. Текущий статус: альфа-версия (~77% готовности). Основные проблемы: недоработано видеоядро (RTSP, ONVIF, запись), нестабильная структура, отсутствие многих функций, критичных для реального использования.

Данный план разработан для одного разработчика (full-stack, C++, Kotlin, DevOps). Цель: получить стабильную, функциональную систему, готовую к пилотному внедрению.

Принципы выполнения
Итеративность: каждый этап даёт работающий инкремент.

Приоритеты: сначала критическое ядро, потом улучшения.

Автоматизация: скрипты, Docker, CI для уменьшения рутины.

Документирование: вести документацию по ходу.

Этап 0: Подготовка и инфраструктура (1–2 недели)
Цель: Создать удобное окружение для разработки, настроить структуру репозитория, CI, управление конфигурацией.

cpp
#include "onvif_wrapper.h"
#include <onvifclient.h>
#include <iostream>
#include <cstring>
#include <vector>

using namespace std;

extern "C" {

StringArray discover_cameras(int timeout_ms) {
    StringArray result = {nullptr, 0};
    try {
        ONVIFClient client;
        auto devices = client.discoverDevices(timeout_ms);
        if (!devices.empty()) {
            result.count = devices.size();
            result.urls = new char*[result.count];
            for (size_t i = 0; i < devices.size(); ++i) {
                string addr = devices[i].getXAddrs()[0];
                result.urls[i] = new char[addr.length() + 1];
                strcpy(result.urls[i], addr.c_str());
            }
        }
    } catch (const exception& e) {
        cerr << "ONVIF discovery error: " << e.what() << endl;
    }
    return result;
}

char* get_stream_uri(const char* ip, const char* username, const char* password) {
    try {
        ONVIFClient client(ip, username, password);
        auto profiles = client.getProfiles();
        if (profiles.empty()) return nullptr;
        string token = profiles[0].getToken();
        string uri = client.getStreamUri(token);
        char* result = new char[uri.length() + 1];
        strcpy(result, uri.c_str());
        return result;
    } catch (const exception& e) {
        cerr << "ONVIF getStreamUri error: " << e.what() << endl;
        return nullptr;
    }
}

void free_string_array(StringArray* arr) {
    if (arr->urls) {
        for (int i = 0; i < arr->count; ++i) {
            delete[] arr->urls[i];
        }
        delete[] arr->urls;
        arr->urls = nullptr;
        arr->count = 0;
    }
}

} // extern "C"

package com.ipcss.onvif

import kotlinx.cinterop.*

actual class OnvifDiscover {
    actual fun discoverCameras(timeoutMs: Int): List<String> {
        val result = mutableListOf<String>()
        val discoveryResult = discover_cameras(timeoutMs) ?: return result
        try {
            for (i in 0 until discoveryResult.count) {
                val url = discoveryResult.urls!![i]?.toKString()
                if (url != null) result.add(url)
            }
        } finally {
            free_string_array(discoveryResult.ptr)
        }
        return result
    }

    actual fun getStreamUri(ip: String, username: String?, password: String?): String? {
        val cIp = ip.cstr
        val cUser = username?.cstr ?: null
        val cPass = password?.cstr ?: null
        val uriPtr = get_stream_uri(cIp, cUser, cPass)
        return uriPtr?.toKString()?.also { platform.posix.free(uriPtr) }
    }
}
1.1.7. Интеграция с DiscoverCamerasUseCase
В shared/src/commonMain/kotlin/com/ipcss/domain/DiscoverCamerasUseCase.kt:

kotlin
class DiscoverCamerasUseCase(
    private val onvifDiscover: OnvifDiscover,
    private val cameraRepository: CameraRepository
) {
    suspend operator fun invoke(): List<Camera> = withContext(Dispatchers.IO) {
        val discoveredUrls = onvifDiscover.discoverCameras(3000)
        val cameras = discoveredUrls.mapNotNull { url ->
            val ip = extractIpFromUrl(url)
            val rtspUrl = onvifDiscover.getStreamUri(ip, null, null)
            if (rtspUrl != null) {
                Camera(
                    id = generateId(),
                    name = "Camera at $ip",
                    ip = ip,
                    rtspUrl = rtspUrl,
                    username = null,
                    password = null
                )
            } else null
        }
        cameraRepository.addCameras(cameras)
        cameras
    }

    private fun extractIpFromUrl(url: String): String {
        // Пример: http://192.168.1.100:8080/onvif/device_service
        return url.substringAfter("//").substringBefore(":").substringBefore("/")
    }
}

package com.ipcss.routes

import io.ktor.http.*
import io.ktor.server.application.*
import io.ktor.server.request.*
import io.ktor.server.response.*
import io.ktor.server.routing.*
import java.util.Base64

fun Route.streamAuth() {
    get("/api/auth/stream") {
        val authHeader = call.request.headers["Authorization"]
            ?: return@get call.respond(HttpStatusCode.Unauthorized, "Missing Authorization header")

        if (!authHeader.startsWith("Basic ")) {
            return@get call.respond(HttpStatusCode.Unauthorized, "Invalid auth scheme")
        }

        val base64 = authHeader.substring(6)
        val decoded = try {
            String(Base64.getDecoder().decode(base64))
        } catch (e: IllegalArgumentException) {
            return@get call.respond(HttpStatusCode.Unauthorized, "Invalid base64")
        }

        val parts = decoded.split(":", limit = 2)
        if (parts.size != 2) {
            return@get call.respond(HttpStatusCode.Unauthorized, "Invalid credentials format")
        }

        val password = parts[1] // JWT токен

        val userId = validateJwtAndGetUserId(password)
        if (userId == null) {
            return@get call.respond(HttpStatusCode.Unauthorized, "Invalid token")
        }

        val path = call.request.queryParameters["path"]
        val action = call.request.queryParameters["action"]

        if (path != null && action != null) {
            val hasPermission = checkCameraPermission(userId, path, action)
            if (!hasPermission) {
                return@get call.respond(HttpStatusCode.Forbidden, "Access denied")
            }
        }

        call.respond(HttpStatusCode.OK)
    }
}

import React, { useRef, useEffect } from 'react';
import Hls from 'hls.js';

interface HLSPlayerProps {
  src: string;
  autoPlay?: boolean;
  controls?: boolean;
}

const HLSPlayer: React.FC<HLSPlayerProps> = ({ src, autoPlay = true, controls = true }) => {
  const videoRef = useRef<HTMLVideoElement>(null);

  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    if (video.canPlayType('application/vnd.apple.mpegurl')) {
      video.src = src;
    } else if (Hls.isSupported()) {
      const hls = new Hls();
      hls.loadSource(src);
      hls.attachMedia(video);
      return () => hls.destroy();
    } else {

1.4.3. Индексация записанных файлов
Создать services/RecordingScanner.kt:

kotlin
package com.ipcss.services

import com.ipcss.db.RecordingRepository
import kotlinx.coroutines.*
import java.io.File
import java.time.LocalDateTime
import java.time.format.DateTimeFormatter
import java.util.*

class RecordingScanner(
    private val recordingsDir: String,
    private val repository: RecordingRepository
) {
    private val scope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private val dateFormatter = DateTimeFormatter.ofPattern("yyyy-MM-dd_HH-mm-ss")

    fun start() {
        scope.launch {
            while (true) {
                delay(60_000) // каждую минуту
                scan()
            }
        }
    }

    private suspend fun scan() {
        val baseDir = File(recordingsDir)
        if (!baseDir.exists()) return

        val allFiles = baseDir.walk().filter { it.extension == "mp4" }.toList()
        val existingPaths = repository.getAllFilePaths()

        val newFiles = allFiles.filter { it.absolutePath !in existingPaths }

        for (file in newFiles) {
            val cameraId = extractCameraIdFromPath(file.absolutePath)
            val startTime = extractStartTimeFromFileName(file.name)
            repository.addRecording(
                id = UUID.randomUUID().toString(),
                cameraId = cameraId,
                filePath = file.absolutePath,
                startTime = startTime,
                endTime = null
            )
        }
    }

    private fun extractCameraIdFromPath(path: String): String {
        val parts = path.split(File.separator)
        return parts[parts.size - 2]
    }

    private fun extractStartTimeFromFileName(fileName: String): LocalDateTime {
        val baseName = fileName.removeSuffix(".mp4")
        return LocalDateTime.parse(baseName, dateFormatter)
    }
}

4.4.3. Контрольные суммы
После завершения записи вычислить SHA-256 и сохранить в recordings. Периодически проверять целостность.

kotlin
fun calculateSha256(file: File): String {
    val digest = MessageDigest.getInstance("SHA-256")
    file.inputStream().use { fis ->
        val buffer = ByteArray(8192)
        var read = fis.read(buffer)
        while (read > 0) {
            digest.update(buffer, 0, read)
            read = fis.read(buffer)
        }
    }
    return digest.digest().joinToString("") { "%02x".format(it) }
}

fun exportToAxf(recordingId: String): File {

5.3. Создание установочных пакетов
5.3.1. Docker-образ
В корне Dockerfile:

dockerfile
FROM openjdk:17-jdk-slim
COPY platforms/server-jvm/api/build/libs/api-all.jar /app.jar
COPY mediamtx.yml /mediamtx.yml
EXPOSE 8080 8554 8888 8889
CMD ["java", "-jar", "/app.jar"]
5.3.2. Deb-пакет
bash
jpackage --name ipcss --input platforms/server-jvm/api/build/libs --main-jar api-all.jar --type deb --linux-package-name ipcss --linux-app-category video
5.3.3. Windows-инсталлятор
Аналогично с --type exe или Inno Setup.
